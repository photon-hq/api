```python Python
import os
```

```rust Rust
use photon_ai_api::{Credential, PhotonClientBuilder, SecretString};

let token = std::env::var("PHOTON_API_TOKEN")?;
let client = PhotonClientBuilder::new()
    .credential("accountServiceKey", Credential::Bearer(SecretString::from(token)))
    .build()?;
let result = client
    .preview_organization_invitation(
        &serde_json::from_value(serde_json::json!({
            "token": "token",
        }))?,
    )
    .await
    .map_err(|error| format!("{error:?}"))?
    .into_inner();
```

```go Go
import (
	"context"
	"encoding/json"
	"os"

	photonapi "github.com/photon-hq/api/packages/go"
)

ctx := context.Background()
client := photonapi.NewClient(photonapi.WithAPIKey(os.Getenv("PHOTON_API_TOKEN")))
var body photonapi.PreviewOrganizationInvitationRequest
if err := json.Unmarshal([]byte(`{
	"token": "token"
}`), &body); err != nil {
	return err
}
result, err := client.Organizations.PreviewOrganizationInvitation(ctx, &photonapi.PreviewOrganizationInvitationParams{
	Body: body,
})
if err != nil {
	return err
}
_ = result
```
