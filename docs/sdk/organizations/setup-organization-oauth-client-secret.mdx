`PUT /v1/organizations/{organizationId}/oauth/clients/{clientId}/secret`

Mints the app's first secret; once it has one, rotate it instead. Retain the same UUID Idempotency-Key when retrying: only the delivering request returns plaintext.

```python Python
import os

from photon_api import Photon
from photon_api.rpc_generated import SetupOrganizationOauthClientSecretInput

photon = Photon(headers={"Authorization": f"Bearer {os.environ['PHOTON_API_TOKEN']}"})
result = photon.organizations.setup_organization_oauth_client_secret(
    SetupOrganizationOauthClientSecretInput.model_validate({
        "path": {
            "organizationId": "organizationId",
            "clientId": "clientId",
        },
        "headers": {
            "idempotency-key": "idempotency-key",
        },
    })
)
```

```rust Rust
use photon_ai_api::{Credential, PhotonClientBuilder, SecretString};

let token = std::env::var("PHOTON_API_TOKEN")?;
let client = PhotonClientBuilder::new()
    .credential("accountServiceKey", Credential::Bearer(SecretString::from(token)))
    .build()?;
let result = client
    .setup_organization_oauth_client_secret(
        "organizationId".to_owned(),
        "clientId".to_owned(),
        "idempotency-key".to_owned(),
    )
    .await
    .map_err(|error| format!("{error:?}"))?
    .into_inner();
```

```go Go
import (

ctx := context.Background()

result, err := client.Organizations.SetupOrganizationOauthClientSecret(ctx, &photonapi.SetupOrganizationOauthClientSecretParams{
	OrganizationID: "organizationId",
	ClientID: "clientId",
	IdempotencyKey: "idempotency-key",
})
if err != nil {
	return err
}
_ = result
```
