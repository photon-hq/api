```rust Rust
use photon_ai_api::{Credential, PhotonClientBuilder, SecretString};

let token = std::env::var("PHOTON_API_TOKEN")?;
let client = PhotonClientBuilder::new()

    .build()?;
let result = client
    .create_service_identity(

        "organizationId".to_owned(),

            "name": "name",
            "permissions": [
                "permissions",
            ],
        }))?,
    )
    .await
    .map_err(|error| format!("{error:?}"))?
    .into_inner();
```

```go Go
import (
	"context"
	"encoding/json"
	"os"

	photonapi "github.com/photon-hq/api/packages/go"
)

ctx := context.Background()
client := photonapi.NewClient(photonapi.WithAPIKey(os.Getenv("PHOTON_API_TOKEN")))
var body photonapi.CreateServiceIdentityRequest
if err := json.Unmarshal([]byte(`{
	"name": "name",
	"permissions": [
		"permissions"
	]
}`), &body); err != nil {
	return err
}
result, err := client.Organizations.CreateServiceIdentity(ctx, &photonapi.CreateServiceIdentityParams{
	IdempotencyKey: "idempotencyKey",
	OrganizationID: "organizationId",
	Body: body,
})
if err != nil {
	return err
}
_ = result
```
