```go Go
import (
	"context"
	"encoding/json"
	"os"

	photonapi "github.com/photon-hq/api/packages/go"
)

ctx := context.Background()
client := photonapi.NewClient(photonapi.WithAPIKey(os.Getenv("PHOTON_API_TOKEN")))
var body photonapi.AcceptOrganizationInvitationRequest
if err := json.Unmarshal([]byte(`{
	"token": "token"
}`), &body); err != nil {
	return err
}
result, err := client.Organizations.AcceptOrganizationInvitation(ctx, &photonapi.AcceptOrganizationInvitationParams{
	IdempotencyKey: "idempotencyKey",
	Body: body,
})
if err != nil {
	return err
}
_ = result
```
