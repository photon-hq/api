<RequestExample>
```typescript TypeScript
import { Photon } from "@photon-ai/api";

const photon = new Photon({
  headers: { Authorization: `Bearer ${process.env.PHOTON_API_TOKEN}` },
});
const result = await photon.organizations.getOrganization({
  path: {
    organizationId: "organizationId",
  },
});
```

```python Python
import os

from photon_api import Photon
from photon_api.rpc_generated import GetOrganizationInput

photon = Photon(headers={"Authorization": f"Bearer {os.environ['PHOTON_API_TOKEN']}"})
result = photon.organizations.get_organization(
    GetOrganizationInput.model_validate({
        "path": {
            "organizationId": "organizationId",
        },
    })
)
```

```rust Rust
use photon_ai_api::{Credential, PhotonClientBuilder, SecretString};

let token = std::env::var("PHOTON_API_TOKEN")?;
let client = PhotonClientBuilder::new()
    .credential("accountServiceKey", Credential::Bearer(SecretString::from(token)))
    .build()?;
let result = client
    .get_organization("organizationId".to_owned())
    .await
    .map_err(|error| format!("{error:?}"))?
    .into_inner();
```

```go Go
import (
	"context"
	"os"

	photonapi "github.com/photon-hq/api/packages/go"
)

ctx := context.Background()
client := photonapi.NewClient(photonapi.WithAPIKey(os.Getenv("PHOTON_API_TOKEN")))
result, err := client.Organizations.GetOrganization(ctx, &photonapi.GetOrganizationParams{
	OrganizationID: "organizationId",
})
if err != nil {
	return err
}
_ = result
```
