import (

ctx := context.Background()

var body photonapi.CommitOrganizationAvatarRequest
if err := json.Unmarshal([]byte(`{
	"uploadId": "uploadId",
	"expectedVersion": "expectedVersion"
}`), &body); err != nil {
	return err
}
result, err := client.Organizations.CommitOrganizationAvatar(ctx, &photonapi.CommitOrganizationAvatarParams{
	IdempotencyKey: "idempotencyKey",
	OrganizationID: "organizationId",
	Body: body,
})
if err != nil {
	return err
}
_ = result
