```python Python
import os

from photon_api import Photon
from photon_api.rpc_generated import GetOrganizationOauthClientInput

photon = Photon(headers={"Authorization": f"Bearer {os.environ['PHOTON_API_TOKEN']}"})
result = photon.organizations.get_organization_oauth_client(
    GetOrganizationOauthClientInput.model_validate({
        "path": {
            "organizationId": "organizationId",
            "clientId": "clientId",
        },
    })
)
```

```rust Rust
use photon_ai_api::{Credential, PhotonClientBuilder, SecretString};

let token = std::env::var("PHOTON_API_TOKEN")?;
let client = PhotonClientBuilder::new()
    .credential("accountServiceKey", Credential::Bearer(SecretString::from(token)))
    .build()?;
let result = client
    .get_organization_oauth_client("organizationId".to_owned(), "clientId".to_owned())
    .await
    .map_err(|error| format!("{error:?}"))?
    .into_inner();
```

```go Go
import (
	"context"
	"os"

	photonapi "github.com/photon-hq/api/packages/go"
)

ctx := context.Background()
client := photonapi.NewClient(photonapi.WithAPIKey(os.Getenv("PHOTON_API_TOKEN")))
result, err := client.Organizations.GetOrganizationOauthClient(ctx, &photonapi.GetOrganizationOauthClientParams{
	OrganizationID: "organizationId",
	ClientID: "clientId",
})
if err != nil {
	return err
}
_ = result
```
