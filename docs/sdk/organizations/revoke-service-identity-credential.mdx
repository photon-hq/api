```go Go
import (
	"context"
	"encoding/json"
	"os"

	photonapi "github.com/photon-hq/api/packages/go"
)

ctx := context.Background()
client := photonapi.NewClient(photonapi.WithAPIKey(os.Getenv("PHOTON_API_TOKEN")))
var body photonapi.RevokeServiceIdentityCredentialRequest
if err := json.Unmarshal([]byte(`{
	"expectedVersion": "expectedVersion"
}`), &body); err != nil {
	return err
}
result, err := client.Organizations.RevokeServiceIdentityCredential(ctx, &photonapi.RevokeServiceIdentityCredentialParams{
	IdempotencyKey: "idempotencyKey",
	OrganizationID: "organizationId",
	ServiceIdentityID: "serviceIdentityId",
	CredentialID: "credentialId",
	Body: body,
})
if err != nil {
	return err
}
_ = result
```
