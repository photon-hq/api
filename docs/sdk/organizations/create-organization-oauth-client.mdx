import (

if err := json.Unmarshal([]byte(`{
	"name": "name",
	"redirectUris": [

		}
	],
	"scopes": []
}`), &body); err != nil {
	return err
}
result, err := client.Organizations.CreateOrganizationOauthClient(ctx, &photonapi.CreateOrganizationOauthClientParams{
	OrganizationID: "organizationId",
	IdempotencyKey: "idempotency-key",
	Body: body,
})
if err != nil {
	return err
}
_ = result
