```python Python
import os

from photon_api import Photon

        "path": {
            "organizationId": "organizationId",

        "headers": {

        },
        "body": {
            "expectedVersion": "expectedVersion",
        },
    })
)
```

```rust Rust
use photon_ai_api::{Credential, PhotonClientBuilder, SecretString};

let token = std::env::var("PHOTON_API_TOKEN")?;

        "organizationId".to_owned(),

    .map_err(|error| format!("{error:?}"))?
    .into_inner();
```

```go Go
import (
	"context"
	"encoding/json"
	"os"

	photonapi "github.com/photon-hq/api/packages/go"
)

ctx := context.Background()
client := photonapi.NewClient(photonapi.WithAPIKey(os.Getenv("PHOTON_API_TOKEN")))
var body photonapi.UpdateServiceIdentityRequest
if err := json.Unmarshal([]byte(`{
	"expectedVersion": "expectedVersion"
}`), &body); err != nil {
	return err
}
result, err := client.Organizations.UpdateServiceIdentity(ctx, &photonapi.UpdateServiceIdentityParams{
	IdempotencyKey: "idempotencyKey",
	OrganizationID: "organizationId",
	ServiceIdentityID: "serviceIdentityId",
	Body: body,
})
if err != nil {
	return err
}
_ = result
```
