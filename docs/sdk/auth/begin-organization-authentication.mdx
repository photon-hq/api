`POST /v1/auth/organizations/{orgId}/sso/authenticate`

Returns a URL to authenticate through the selected organization’s current SSO connection. Supply returnTo and open the returned URL to continue the flow. Receiving the URL does not establish an authenticated session.

<RequestExample>
```typescript TypeScript
import { Photon } from "@photon-ai/api";

const photon = new Photon({
  headers: { Authorization: `Bearer ${process.env.PHOTON_API_TOKEN}` },
});
const result = await photon.auth.beginOrganizationAuthentication({
  path: {
    orgId: "orgId",
  },
  body: {
    returnTo: "returnTo",
  },
});
```

```python Python
import os

from photon_api import Photon
from photon_api.rpc_generated import BeginOrganizationAuthenticationInput

photon = Photon(headers={"Authorization": f"Bearer {os.environ['PHOTON_API_TOKEN']}"})
result = photon.auth.begin_organization_authentication(
    BeginOrganizationAuthenticationInput.model_validate({
        "path": {
            "orgId": "orgId",
        },
        "body": {
            "returnTo": "returnTo",
        },
    })
)
```

```rust Rust
use photon_ai_api::{Credential, PhotonClientBuilder, SecretString};

let token = std::env::var("PHOTON_API_TOKEN")?;
let client = PhotonClientBuilder::new()
    .credential("accountServiceKey", Credential::Bearer(SecretString::from(token)))
    .build()?;
let result = client
    .begin_organization_authentication(
        "orgId".to_owned(),
        &serde_json::from_value(serde_json::json!({
            "returnTo": "returnTo",
        }))?,
    )
    .await
    .map_err(|error| format!("{error:?}"))?
    .into_inner();
```
