`POST /v1/auth/organizations/{orgId}/sso/invitation`

Returns an authentication URL for the organization SSO connection associated with the supplied invitation token. Supply token and returnTo. Complete the returned authentication flow; requesting its URL does not itself accept the invitation.

<RequestExample>
```typescript TypeScript
import { Photon } from "@photon-ai/api";

const photon = new Photon({
  headers: { Authorization: `Bearer ${process.env.PHOTON_API_TOKEN}` },
});
const result = await photon.auth.beginInvitationSso({
  path: {
    orgId: "orgId",
  },
  body: {
    token: "token",
    returnTo: "returnTo",
  },
});
```

```python Python
import os

from photon_api import Photon
from photon_api.rpc_generated import BeginInvitationSsoInput

photon = Photon(headers={"Authorization": f"Bearer {os.environ['PHOTON_API_TOKEN']}"})
result = photon.auth.begin_invitation_sso(
    BeginInvitationSsoInput.model_validate({
        "path": {
            "orgId": "orgId",
        },
        "body": {
            "token": "token",
            "returnTo": "returnTo",
        },
    })
)
```

```rust Rust
use photon_ai_api::{Credential, PhotonClientBuilder, SecretString};

let token = std::env::var("PHOTON_API_TOKEN")?;
let client = PhotonClientBuilder::new()
    .credential("accountServiceKey", Credential::Bearer(SecretString::from(token)))
    .build()?;
let result = client
    .begin_invitation_sso(
        "orgId".to_owned(),
        &serde_json::from_value(serde_json::json!({
            "token": "token",
            "returnTo": "returnTo",
        }))?,
    )
    .await
    .map_err(|error| format!("{error:?}"))?
    .into_inner();
```
