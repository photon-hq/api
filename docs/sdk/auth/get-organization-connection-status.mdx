`GET /v1/auth/organizations/{orgId}/connection-status`

Requires current human organization membership. Synchronization status does not attest SSO configuration or completed authorization.


```rust Rust
use photon_ai_api::{Credential, PhotonClientBuilder, SecretString};

let token = std::env::var("PHOTON_API_TOKEN")?;
let client = PhotonClientBuilder::new()
    .credential("accountServiceKey", Credential::Bearer(SecretString::from(token)))
    .build()?;
let result = client
    .get_organization_connection_status("orgId".to_owned())
    .await
    .map_err(|error| format!("{error:?}"))?
    .into_inner();
```
