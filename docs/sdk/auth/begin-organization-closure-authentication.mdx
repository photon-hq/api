`POST /v1/auth/organizations/{orgId}/sso/closure-authenticate`

Returns an authentication URL for the current organization owner to inspect organization closure. Supply returnTo and complete the returned flow. This operation initiates authentication and does not close the organization.

```go Go
import (
	"context"
	"encoding/json"
	"os"

	photonapi "github.com/photon-hq/api/packages/go"
)

ctx := context.Background()
client := photonapi.NewClient(photonapi.WithAPIKey(os.Getenv("PHOTON_API_TOKEN")))
var body photonapi.OrganizationAuthenticationRequest
if err := json.Unmarshal([]byte(`{
	"returnTo": "returnTo"
}`), &body); err != nil {
	return err
}
result, err := client.Auth.BeginOrganizationClosureAuthentication(ctx, &photonapi.BeginOrganizationClosureAuthenticationParams{
	OrgID: "orgId",
	Body: body,
})
if err != nil {
	return err
}
_ = result
```
