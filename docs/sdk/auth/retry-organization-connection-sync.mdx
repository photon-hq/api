`POST /v1/auth/organizations/{orgId}/connection-sync/retry`

Reconciles existing local intent. Takes no body and cannot change membership, roles or authentication policy.

```rust Rust
use photon_ai_api::{Credential, PhotonClientBuilder, SecretString};

let token = std::env::var("PHOTON_API_TOKEN")?;

    .build()?;

    .retry_organization_connection_sync("orgId".to_owned())
    .await
    .map_err(|error| format!("{error:?}"))?
```

```go Go
import (
	"context"
	"os"

	photonapi "github.com/photon-hq/api/packages/go"
)

ctx := context.Background()
client := photonapi.NewClient(photonapi.WithAPIKey(os.Getenv("PHOTON_API_TOKEN")))
result, err := client.Auth.RetryOrganizationConnectionSync(ctx, &photonapi.RetryOrganizationConnectionSyncParams{
	OrgID: "orgId",
})
if err != nil {
	return err
}
_ = result
```
