`DELETE /v1/account/service-keys/{serviceKeyId}`

Revokes the account-owned service key identified by serviceKeyId and returns its revoked metadata. Repeating the revocation is stable. Revocation changes the credential's validity; it does not create a replacement key. Supply the required Idempotency-Key header.

```python Python
import os

from photon_api import Photon

photon = Photon(headers={"Authorization": f"Bearer {os.environ['PHOTON_API_TOKEN']}"})
result = photon.account.revoke_account_service_key(

        "path": {

        },
        "headers": {

        },
```

```rust Rust
use photon_ai_api::{Credential, PhotonClientBuilder, SecretString};

let token = std::env::var("PHOTON_API_TOKEN")?;
let client = PhotonClientBuilder::new()
    .credential("accountServiceKey", Credential::Bearer(SecretString::from(token)))
    .build()?;
let result = client
    .revoke_account_service_key("idempotencyKey".to_owned(), "serviceKeyId".to_owned())
    .await
    .map_err(|error| format!("{error:?}"))?
    .into_inner();
```
