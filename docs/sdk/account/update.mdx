`PATCH /v1/account`

Updates the supplied firstName and lastName fields on the authenticated account and returns the updated profile. Only the documented profile fields can be changed through this endpoint; profile-picture uploads and phone-number verification use their dedicated operations. Supply the required Idempotency-Key header.

```python Python
import os

from photon_api import Photon
from photon_api.rpc_generated import UpdateAccountInput

photon = Photon(headers={"Authorization": f"Bearer {os.environ['PHOTON_API_TOKEN']}"})
result = photon.account.update(
    UpdateAccountInput.model_validate({
        "headers": {
            "Idempotency-Key": "idempotencyKey",
        },
        "body": {},
    })
)
```

```rust Rust
use photon_ai_api::{Credential, PhotonClientBuilder, SecretString};

let token = std::env::var("PHOTON_API_TOKEN")?;
let client = PhotonClientBuilder::new()
    .credential("accountServiceKey", Credential::Bearer(SecretString::from(token)))
    .build()?;
let result = client
    .update_account(
        "idempotencyKey".to_owned(),
        &serde_json::from_value(serde_json::json!({}))?,
    )
    .await
    .map_err(|error| format!("{error:?}"))?
    .into_inner();
```
