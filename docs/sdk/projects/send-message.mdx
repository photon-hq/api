`POST /v1/projects/{id}/messages`

Sends SMS, email, WhatsApp Personal, WhatsApp Business, iMessage, or Telegram text and image messages. A shared iMessage line sends only in direct Spaces, only to a peer with an active Assignment who has messaged it since that Assignment. SMS and MMS require current Platforms authorization for the sender and every recipient; campaign eligibility is required unless every recipient is in-network.

```python Python
import os

from photon_api import Photon
from photon_api.rpc_generated import SendMessageInput

photon = Photon(headers={"Authorization": f"Bearer {os.environ['PHOTON_API_TOKEN']}"})
result = photon.projects.send_message(
    SendMessageInput.model_validate({
        "path": {
            "id": "id",
        },
        "headers": {
            "idempotency-key": "idempotency-key",
        },
        "body": {
            "content": {
                "type": "text",
                "format": "format",
                "text": "text",
            },
            "senderId": "senderId",
            "spaceId": "spaceId",
        },
    })
)
```

```rust Rust
use photon_ai_api::{Credential, PhotonClientBuilder, SecretString};

let token = std::env::var("PHOTON_API_TOKEN")?;
let client = PhotonClientBuilder::new()
    .credential("accountServiceKey", Credential::Bearer(SecretString::from(token)))
    .build()?;
let result = client
    .send_message(
        "id".to_owned(),
        "idempotency-key".to_owned(),
        &serde_json::from_value(serde_json::json!({
            "content": {
                "type": "text",
                "format": "format",
                "text": "text",
            },
            "senderId": "senderId",
            "spaceId": "spaceId",
        }))?,
    )
    .await
    .map_err(|error| format!("{error:?}"))?
    .into_inner();
```
