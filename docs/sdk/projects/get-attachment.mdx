`GET /v1/projects/{id}/attachments/{attachmentId}`

Returns an Attachment's metadata. Use the content endpoint to download its bytes.

<RequestExample>
```typescript TypeScript
import { Photon } from "@photon-ai/api";

const photon = new Photon({
  headers: { Authorization: `Bearer ${process.env.PHOTON_API_TOKEN}` },
});
const result = await photon.projects.getAttachment({
  path: {
    attachmentId: "attachmentId",
    id: "id",
  },
});
```

```python Python
import os

from photon_api import Photon
from photon_api.rpc_generated import GetAttachmentInput

photon = Photon(headers={"Authorization": f"Bearer {os.environ['PHOTON_API_TOKEN']}"})
result = photon.projects.get_attachment(
    GetAttachmentInput.model_validate({
        "path": {
            "attachmentId": "attachmentId",
            "id": "id",
        },
    })
)
```

```rust Rust
use photon_ai_api::{Credential, PhotonClientBuilder, SecretString};

let token = std::env::var("PHOTON_API_TOKEN")?;
let client = PhotonClientBuilder::new()
    .credential("accountServiceKey", Credential::Bearer(SecretString::from(token)))
    .build()?;
let result = client
    .get_attachment("attachmentId".to_owned(), "id".to_owned())
    .await
    .map_err(|error| format!("{error:?}"))?
    .into_inner();
```

</RequestExample>
