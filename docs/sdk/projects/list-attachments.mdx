`GET /v1/projects/{id}/attachments`

Lists the Project's Attachment metadata, with optional time filters.

```typescript TypeScript
import { Photon } from "@photon-ai/api";

const photon = new Photon({
```

```python Python
import os

from photon_api import Photon
from photon_api.rpc_generated import ListAttachmentsInput

photon = Photon(headers={"Authorization": f"Bearer {os.environ['PHOTON_API_TOKEN']}"})
result = photon.projects.list_attachments(
    ListAttachmentsInput.model_validate({
        "path": {
            "id": "id",
        },
    })
)
```

```rust Rust
use photon_ai_api::{Credential, PhotonClientBuilder, SecretString};

let token = std::env::var("PHOTON_API_TOKEN")?;
let client = PhotonClientBuilder::new()
    .credential("accountServiceKey", Credential::Bearer(SecretString::from(token)))
    .build()?;
let result = client
    .list_attachments("id".to_owned(), None)
    .await
    .map_err(|error| format!("{error:?}"))?
    .into_inner();
```
