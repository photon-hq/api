`POST /v1/projects/{id}/attachments`

Uploads a file and returns its Attachment once ready to download.

<RequestExample>
```typescript TypeScript
import { Photon } from "@photon-ai/api";

const photon = new Photon({
  headers: { Authorization: `Bearer ${process.env.PHOTON_API_TOKEN}` },
});
const result = await photon.projects.uploadAttachment({
  path: {
    id: "id",
  },
  headers: {
    "content-length": "18",
    "content-type": "text/plain",
    "idempotency-key": "idempotency-key",
  },
  body: new Blob(["Hello from Photon."], { type: "text/plain" }),
});
```

```python Python
import os

from photon_api import Photon
from photon_api.rpc_generated import UploadAttachmentInput

photon = Photon(headers={"Authorization": f"Bearer {os.environ['PHOTON_API_TOKEN']}"})
result = photon.projects.upload_attachment(
    UploadAttachmentInput.model_validate({
        "path": {
            "id": "id",
        },
        "headers": {
            "content-length": "18",
            "content-type": "text/plain",
            "idempotency-key": "idempotency-key",
        },
        "body": b"Hello from Photon.",
    })
)
```

```rust Rust
use photon_ai_api::{Credential, PhotonClientBuilder, SecretString};

let token = std::env::var("PHOTON_API_TOKEN")?;
let client = PhotonClientBuilder::new()
    .credential("accountServiceKey", Credential::Bearer(SecretString::from(token)))
    .build()?;
let result = client
    .upload_attachment(
        "id".to_owned(),
        "18".to_owned(),
        "text/plain".to_owned(),
        "idempotency-key".to_owned(),
        &"Hello from Photon.".into(),
    )
    .await
    .map_err(|error| format!("{error:?}"))?
    .into_inner();
```
