`DELETE /v1/projects/{projectId}`

Starts deletion of the identified project using a credential authorized for project management. Inspect the documented response and use getProjectClosureStatus with the organization and project identifiers to read closure progress. A project API key is not an accepted credential for this operation.

```python Python
import os

from photon_api import Photon
from photon_api.rpc_generated import DeleteProjectInput

photon = Photon(headers={"Authorization": f"Bearer {os.environ['PHOTON_API_TOKEN']}"})
result = photon.projects.delete(
    DeleteProjectInput.model_validate({
        "path": {
            "projectId": "projectId",
        },
        "headers": {
            "Idempotency-Key": "idempotencyKey",
        },
    })
)
```

```rust Rust
use photon_ai_api::{Credential, PhotonClientBuilder, SecretString};

let token = std::env::var("PHOTON_API_TOKEN")?;
let client = PhotonClientBuilder::new()
    .credential("accountServiceKey", Credential::Bearer(SecretString::from(token)))
    .build()?;
let result = client
    .delete_project("idempotencyKey".to_owned(), "projectId".to_owned())
    .await
    .map_err(|error| format!("{error:?}"))?
    .into_inner();
```
