`POST /v1/projects/{projectId}/metrics/query`

Runs read-only SQL over the Project's message_events or call_events table. Get the SQL schema for supported columns, capabilities, and limits.

import (

ctx := context.Background()

var body photonapi.ClickHouseSqlQueryRequest
if err := json.Unmarshal([]byte(`{
	"sql": "sql"
}`), &body); err != nil {
	return err
}
result, err := client.Projects.QueryMessageMetrics(ctx, &photonapi.QueryMessageMetricsParams{
	ProjectID: "pho_prj_70jqmkvtz0mnhnjnn58nhmn8kw",
	Body: body,
})
if err != nil {
	return err
}
_ = result
