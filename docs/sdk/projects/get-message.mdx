`GET /v1/projects/{id}/messages/{messageId}`

Returns one stored Message by canonical Photon Message ID on any supported contract platform. The globally unique Message ID is resolved within the authorized Project. Unsupported platform Messages are outside this collection and return MESSAGE_NOT_FOUND.

const photon = new Photon({

```python Python
import os

from photon_api import Photon
from photon_api.rpc_generated import GetMessageInput

photon = Photon(headers={"Authorization": f"Bearer {os.environ['PHOTON_API_TOKEN']}"})
result = photon.projects.get_message(
    GetMessageInput.model_validate({
        "path": {
            "id": "id",
            "messageId": "messageId",
        },
    })
)
```

```rust Rust
use photon_ai_api::{Credential, PhotonClientBuilder, SecretString};

let token = std::env::var("PHOTON_API_TOKEN")?;
let client = PhotonClientBuilder::new()
    .credential("accountServiceKey", Credential::Bearer(SecretString::from(token)))
    .build()?;
let result = client
    .get_message("id".to_owned(), "messageId".to_owned(), None)
    .await
    .map_err(|error| format!("{error:?}"))?
    .into_inner();
```

```go Go
import (
	"context"
	"os"

	photonapi "github.com/photon-hq/api/packages/go"
)

ctx := context.Background()
client := photonapi.NewClient(photonapi.WithAPIKey(os.Getenv("PHOTON_API_TOKEN")))
result, err := client.Projects.GetMessage(ctx, &photonapi.GetMessageParams{
	ID: "id",
	MessageID: "messageId",
})
if err != nil {
	return err
}
_ = result
```
