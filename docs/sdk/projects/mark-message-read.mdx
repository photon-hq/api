`POST /v1/projects/{id}/messages/{messageId}/read`

Persists a request to mark one incoming Message as read using its receiving account. A 202 means accepted for asynchronous execution, not handled by the provider. Requires Idempotency-Key. The returned action Message tracks its outcome; privacy settings may suppress remote visibility.

```python Python
import os

from photon_api import Photon
from photon_api.rpc_generated import MarkMessageReadInput

photon = Photon(headers={"Authorization": f"Bearer {os.environ['PHOTON_API_TOKEN']}"})
result = photon.projects.mark_message_read(
    MarkMessageReadInput.model_validate({
        "path": {
            "id": "id",
            "messageId": "messageId",
        },
        "headers": {
            "idempotency-key": "idempotency-key",
        },
    })
)
```

```rust Rust
use photon_ai_api::{Credential, PhotonClientBuilder, SecretString};

let token = std::env::var("PHOTON_API_TOKEN")?;
let client = PhotonClientBuilder::new()
    .credential("accountServiceKey", Credential::Bearer(SecretString::from(token)))
    .build()?;
let result = client
    .mark_message_read(
        "id".to_owned(),
        "messageId".to_owned(),
        "idempotency-key".to_owned(),
    )
    .await
    .map_err(|error| format!("{error:?}"))?
    .into_inner();
```

```go Go
import (
	"context"
	"os"

	photonapi "github.com/photon-hq/api/packages/go"

ctx := context.Background()
client := photonapi.NewClient(photonapi.WithAPIKey(os.Getenv("PHOTON_API_TOKEN")))
result, err := client.Projects.MarkMessageRead(ctx, &photonapi.MarkMessageReadParams{
	ID: "id",
	MessageID: "messageId",
	IdempotencyKey: "idempotency-key",
})
if err != nil {
	return err
}
_ = result
```
