`PATCH /v1/projects/{projectId}`

Updates the identified project's name and returns the updated project. The project slug is not a mutable field in this request. Use an authorized account or organization service-identity credential; a project API key is not accepted. Supply the required Idempotency-Key header.

```python Python
import os

from photon_api import Photon
from photon_api.rpc_generated import UpdateProjectInput

photon = Photon(headers={"Authorization": f"Bearer {os.environ['PHOTON_API_TOKEN']}"})
result = photon.projects.update(
    UpdateProjectInput.model_validate({
        "path": {
            "projectId": "projectId",
        },
        "headers": {
            "Idempotency-Key": "idempotencyKey",
        },
        "body": {
            "name": "name",
        },
    })
)
```

```rust Rust
use photon_ai_api::{Credential, PhotonClientBuilder, SecretString};

let token = std::env::var("PHOTON_API_TOKEN")?;
let client = PhotonClientBuilder::new()
    .credential("accountServiceKey", Credential::Bearer(SecretString::from(token)))
    .build()?;
let result = client
    .update_project(
        "idempotencyKey".to_owned(),
        "projectId".to_owned(),
        &serde_json::from_value(serde_json::json!({
            "name": "name",
        }))?,
    )
    .await
    .map_err(|error| format!("{error:?}"))?
    .into_inner();
```
