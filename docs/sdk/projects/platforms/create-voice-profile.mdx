`POST /v1/projects/{id}/platforms/voice/profiles`

Creates a direction-neutral additional Voice profile. The project default must already exist. Requires platforms:write bound to the path project.

```python Python
import os

from photon_api import Photon
from photon_api.rpc_generated import CreateVoiceProfileInput

photon = Photon(headers={"Authorization": f"Bearer {os.environ['PHOTON_API_TOKEN']}"})
result = photon.projects.platforms.create_voice_profile(
    CreateVoiceProfileInput.model_validate({
        "path": {
            "id": "id",
        },
        "body": {
            "name": "name",
            "protocol": "sip",
        },
    })
)
```

```rust Rust
use photon_ai_api::{Credential, PhotonClientBuilder, SecretString};

let token = std::env::var("PHOTON_API_TOKEN")?;
let client = PhotonClientBuilder::new()
    .credential("accountServiceKey", Credential::Bearer(SecretString::from(token)))
    .build()?;
let result = client
    .create_voice_profile(
        "id".to_owned(),
        &serde_json::from_value(serde_json::json!({
            "name": "name",
            "protocol": "sip",
        }))?,
    )
    .await
    .map_err(|error| format!("{error:?}"))?
    .into_inner();
```
